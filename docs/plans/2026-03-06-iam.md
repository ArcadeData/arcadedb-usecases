# IAM Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a fully self-contained `iam/` directory demonstrating ArcadeDB's multi-model capabilities for Identity & Access Management — graph traversal, time-series auditing, vector anomaly detection — runnable with `curl`, Java, and Python (via PostgreSQL wire protocol).

**Architecture:** Self-contained directory per the project conventions. Docker Compose brings up ArcadeDB 26.3.1 with PostgreSQL plugin. A `setup.sh` creates the database and applies SQL files. Seven queries are demonstrated via `queries/queries.sh` (curl), `java/` (Maven fat JAR using `arcadedb-network`), and `python/` (`psycopg` via PG wire protocol).

**Tech Stack:** ArcadeDB 26.3.1, Docker Compose, Maven 3.x, Java 21, `com.arcadedb:arcadedb-network:26.3.1`, Python 3.12+, `psycopg[binary]`, `jq` (for setup/query scripts)

### Task 3: Write the SQL schema

**Files:**
- Create: `iam/sql/01-schema.sql`

**Step 1: Write the schema file**

One statement per line. Note `identityType` instead of `type` (reserved). `recordedAt` instead of `timestamp` (reserved).

```sql
-- Vertex types
CREATE VERTEX TYPE Identity IF NOT EXISTS
CREATE PROPERTY Identity.email IF NOT EXISTS STRING
CREATE PROPERTY Identity.identityType IF NOT EXISTS STRING
CREATE PROPERTY Identity.department IF NOT EXISTS STRING
CREATE PROPERTY Identity.title IF NOT EXISTS STRING
CREATE PROPERTY Identity.access_pattern_vec IF NOT EXISTS LIST
CREATE INDEX IF NOT EXISTS ON Identity (email) UNIQUE
CREATE VERTEX TYPE `Group` IF NOT EXISTS
CREATE PROPERTY `Group`.name IF NOT EXISTS STRING
CREATE PROPERTY `Group`.description IF NOT EXISTS STRING
CREATE INDEX IF NOT EXISTS ON `Group` (name) UNIQUE
CREATE VERTEX TYPE Role IF NOT EXISTS
CREATE PROPERTY Role.name IF NOT EXISTS STRING
CREATE PROPERTY Role.description IF NOT EXISTS STRING
CREATE INDEX IF NOT EXISTS ON Role (name) UNIQUE
CREATE VERTEX TYPE Permission IF NOT EXISTS
CREATE PROPERTY Permission.action IF NOT EXISTS STRING
CREATE VERTEX TYPE Resource IF NOT EXISTS
CREATE PROPERTY Resource.name IF NOT EXISTS STRING
CREATE PROPERTY Resource.classification IF NOT EXISTS STRING
CREATE PROPERTY Resource.data_sensitivity IF NOT EXISTS STRING
CREATE PROPERTY Resource.compliance_scope IF NOT EXISTS STRING
CREATE INDEX IF NOT EXISTS ON Resource (name) UNIQUE
CREATE VERTEX TYPE Policy IF NOT EXISTS
CREATE PROPERTY Policy.name IF NOT EXISTS STRING
CREATE PROPERTY Policy.policyType IF NOT EXISTS STRING
CREATE PROPERTY Policy.description IF NOT EXISTS STRING
CREATE INDEX IF NOT EXISTS ON Policy (name) UNIQUE
-- Edge types
CREATE EDGE TYPE MEMBER_OF IF NOT EXISTS
CREATE EDGE TYPE HAS_ROLE IF NOT EXISTS
CREATE EDGE TYPE GRANTS IF NOT EXISTS
CREATE EDGE TYPE APPLIES_TO IF NOT EXISTS
CREATE EDGE TYPE INHERITS_FROM IF NOT EXISTS
CREATE EDGE TYPE GOVERNED_BY IF NOT EXISTS
-- Document type for time-series access logs
CREATE DOCUMENT TYPE AccessLog IF NOT EXISTS
CREATE PROPERTY AccessLog.identityEmail IF NOT EXISTS STRING
CREATE PROPERTY AccessLog.resourceName IF NOT EXISTS STRING
CREATE PROPERTY AccessLog.action IF NOT EXISTS STRING
CREATE PROPERTY AccessLog.source_ip IF NOT EXISTS STRING
CREATE PROPERTY AccessLog.recordedAt IF NOT EXISTS DATETIME
-- Vector index for behavioral anomaly detection
CREATE INDEX IF NOT EXISTS ON Identity (access_pattern_vec) LSM_VECTOR METADATA { dimensions: 8, similarity: 'COSINE' }
```

> **Note:** `Group` may be a reserved word in ArcadeDB — backtick-quote it in CREATE statements. Test during setup; if backticks cause issues, rename to `AccessGroup`.

### Task 5: Write setup.sh

**Files:**
- Create: `iam/setup.sh`

**Step 1: Write the script**

Copy from recommendation-engine, change `DB_NAME="IAM"`.

```bash
#!/usr/bin/env bash
set -euo pipefail

ARCADEDB_URL="${ARCADEDB_URL:-http://localhost:2480}"
ARCADEDB_USER="${ARCADEDB_USER:-root}"
ARCADEDB_PASS="${ARCADEDB_PASS:-arcadedb}"
DB_NAME="IAM"

# ── Wait for ArcadeDB ─────────────────────────────────────────────────────────
echo "Waiting for ArcadeDB at ${ARCADEDB_URL}..."
until curl -sf -u "${ARCADEDB_USER}:${ARCADEDB_PASS}" \
    "${ARCADEDB_URL}/api/v1/ready" > /dev/null 2>&1; do
  sleep 2
done
echo "ArcadeDB is ready."

# ── Create database ───────────────────────────────────────────────────────────
echo "Creating database ${DB_NAME}..."
curl -sf -u "${ARCADEDB_USER}:${ARCADEDB_PASS}" \
  -X POST "${ARCADEDB_URL}/api/v1/server" \
  -H "Content-Type: application/json" \
  -d "{\"command\": \"create database ${DB_NAME}\"}" > /dev/null || true
echo "Database ready."

# ── Helper: send one SQL statement ───────────────────────────────────────────
send_sql() {
  local stmt="$1"
  jq -cn --arg cmd "$stmt" '{"language":"sql","command":$cmd}' \
    | curl -sf -u "${ARCADEDB_USER}:${ARCADEDB_PASS}" \
        -X POST "${ARCADEDB_URL}/api/v1/command/${DB_NAME}" \
        -H "Content-Type: application/json" \
        -d @- > /dev/null
}

# ── Apply a SQL file (one statement per line) ─────────────────────────────────
apply_file() {
  local file="$1"
  echo "Applying ${file}..."
  while IFS= read -r line || [[ -n "$line" ]]; do
    # skip blank lines and SQL comments
    [[ -z "${line//[[:space:]]/}" || "$line" =~ ^[[:space:]]*-- ]] && continue
    send_sql "${line%%;}"
  done < "$file"
  echo "Done: ${file}"
}

apply_file "sql/01-schema.sql"
apply_file "sql/02-data.sql"

echo ""
echo "Setup complete. ${DB_NAME} is ready."
```

```bash
#!/usr/bin/env bash
# IAM — all seven query patterns via curl
# Prerequisites: ArcadeDB running, setup.sh already executed, jq installed
# Usage: ./queries/queries.sh

set -euo pipefail

ARCADEDB_URL="${ARCADEDB_URL:-http://localhost:2480}"
ARCADEDB_USER="${ARCADEDB_USER:-root}"
ARCADEDB_PASS="${ARCADEDB_PASS:-arcadedb}"
AUTH="${ARCADEDB_USER}:${ARCADEDB_PASS}"
DB="IAM"
QUERY_URL="${ARCADEDB_URL}/api/v1/query/${DB}"

query() {
  local lang="$1" cmd="$2"
  jq -cn --arg l "$lang" --arg c "$cmd" '{"language":$l,"command":$c}' \
    | curl -sf -u "$AUTH" -X POST "$QUERY_URL" \
        -H "Content-Type: application/json" -d @- \
    | jq '.result'
}

# ─────────────────────────────────────────────────────────────────────────────
echo "=== Query 1: Permission Resolution ==="
echo "Discover all resources alice@company.com can access through any chain."
echo ""
query "cypher" "
MATCH (u:Identity {email: 'alice@company.com'})
      -[:MEMBER_OF*1..3]->(g:Group)
      -[:HAS_ROLE]->(r:Role)
      -[:GRANTS]->(p:Permission)
      -[:APPLIES_TO]->(res:Resource)
RETURN res.name AS resource, p.action AS action,
       r.name AS via_role, g.name AS via_group
ORDER BY resource, action
"

# ─────────────────────────────────────────────────────────────────────────────
echo ""
echo "=== Query 2: Shadow Admin Detection ==="
echo "Find contractors/service accounts with admin on critical resources."
echo ""
query "cypher" "
MATCH (u:Identity)
      -[:MEMBER_OF*1..5]->(g:Group)
      -[:HAS_ROLE]->(r:Role)
      -[:GRANTS]->(p:Permission {action: 'admin'})
      -[:APPLIES_TO]->(res:Resource {classification: 'critical'})
WHERE u.identityType IN ['contractor', 'service_account']
RETURN u.email AS identity, u.identityType AS identity_type,
       res.name AS critical_resource, r.name AS via_role
ORDER BY identity
"

# ─────────────────────────────────────────────────────────────────────────────
echo ""
echo "=== Query 3: SOX Compliance Audit ==="
echo "Track access to SOX-governed resources with policy lineage."
echo ""
query "sql" "
SELECT identity, action, resource, policy, recordedAt, source_ip
FROM (
  MATCH {type: Resource, as: res}
        .out('GOVERNED_BY'){as: pol, where: (name = 'SOX-Compliance')}
  RETURN res.name AS resource, pol.name AS policy
) AS gov
JOIN AccessLog AS al ON al.resourceName = gov.resource
WHERE al.recordedAt > '2025-12-01 00:00:00'
ORDER BY al.recordedAt DESC
"

# Note: If JOIN doesn't work, fall back to two-step approach:
# Step 1: Get SOX-governed resource names
# Step 2: Filter AccessLog by those names

# ─────────────────────────────────────────────────────────────────────────────
echo ""
echo "=== Query 4: Separation of Duties Violation ==="
echo "Find users who can both approve AND execute on the same resource."
echo ""
query "cypher" "
MATCH (u:Identity)
      -[:MEMBER_OF*1..3]->(g1:Group)
      -[:HAS_ROLE]->(r1:Role)
      -[:GRANTS]->(p1:Permission {action: 'approve'})
      -[:APPLIES_TO]->(res:Resource),
      (u)-[:MEMBER_OF*1..3]->(g2:Group)
      -[:HAS_ROLE]->(r2:Role)
      -[:GRANTS]->(p2:Permission {action: 'execute'})
      -[:APPLIES_TO]->(res)
RETURN u.email AS identity, res.name AS resource,
       r1.name AS approve_role, r2.name AS execute_role
"

# ─────────────────────────────────────────────────────────────────────────────
echo ""
echo "=== Query 5: Dormant Access Detection ==="
echo "Find identities with permissions but no access in the last 90 days."
echo ""

echo "--- Step 1: Identities with granted permissions ---"
query "sql" "
SELECT DISTINCT identity, resource
FROM (
  MATCH {type: Identity, as: u}
        .out('MEMBER_OF'){while: (\$depth < 3)}
        .out('HAS_ROLE'){}
        .out('GRANTS'){}
        .out('APPLIES_TO'){as: res}
  RETURN u.email AS identity, res.name AS resource
)
ORDER BY identity
"

echo ""
echo "--- Step 2: Identities with recent access (last 90 days) ---"
query "sql" "
SELECT DISTINCT identityEmail
FROM AccessLog
WHERE recordedAt > '2025-12-06 00:00:00'
ORDER BY identityEmail
"

echo ""
echo "Dormant identities = Step 1 minus Step 2 (frank@company.com, bob@company.com, svc-backup@company.com)"

# ─────────────────────────────────────────────────────────────────────────────
echo ""
echo "=== Query 6: Behavioral Anomaly Detection ==="
echo "Rank employees by how similar their access pattern is to a 'normal' session."
echo ""
query "sql" "
SELECT email, department, identityType
FROM Identity
WHERE identityType = 'employee'
ORDER BY vectorNeighbors('Identity[access_pattern_vec]', [0.5, 0.5, 0.3, 0.1, 0.2, 0.1, 0.1, 0.1], 10) DESC
LIMIT 10
"

# ─────────────────────────────────────────────────────────────────────────────
echo ""
echo "=== Query 7: Impact Analysis (What-If) ==="
echo "What happens if we remove the Platform-Admins group?"
echo ""
query "cypher" "
MATCH (u:Identity)
      -[:MEMBER_OF*1..3]->(target:Group {name: 'Platform-Admins'})
      -[:HAS_ROLE]->(r:Role)
      -[:GRANTS]->(p:Permission)
      -[:APPLIES_TO]->(res:Resource)
RETURN u.email AS identity, r.name AS role,
       p.action AS action, res.name AS resource
ORDER BY identity, resource
"
```

> **Implementation notes:**
> - Query 3 (SOX Audit) uses a SQL MATCH + JOIN. If ArcadeDB 26.3.1 doesn't support JOIN with MATCH subquery, refactor to a two-step approach: first query SOX-governed resources, then filter AccessLog by those names.
> - Query 5 (Dormant Access) is split into two steps because `NOT IN (SELECT ...)` with MATCH subquery may not work. The client computes the set difference.
> - Cypher variable-length paths `[:MEMBER_OF*1..5]` must be tested — if ArcadeDB's Cypher doesn't support this, fall back to SQL MATCH `{while: ($depth < 5)}`.

### Task 11: Write CI workflow

**Files:**
- Create: `.github/workflows/iam.yml`

**Step 1: Write the workflow**

Three-runner matrix: `[curl, java, python]`. Based on the supply-chain workflow pattern (which has `[curl, java, js]`).

```yaml
name: IAM CI

on:
  push:
    paths:
      - iam/**
      - .github/workflows/iam.yml
  pull_request:
    paths:
      - iam/**
      - .github/workflows/iam.yml

jobs:
  test:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    permissions:
      contents: read
    strategy:
      fail-fast: false
      matrix:
        runner: [curl, java, python]

    env:
      ARCADEDB_URL: http://localhost:2480
      ARCADEDB_USER: root
      ARCADEDB_PASS: arcadedb

    steps:
      - name: Checkout
        uses: actions/checkout@de0fac2e4500dabe0009e67214ff5f5447ce83dd # v6.0.2
        with:
          fetch-depth: 1

      - name: Set up Java
        if: matrix.runner == 'java'
        uses: actions/setup-java@be666c2fcd27ec809703dec50e508c2fdc7f6654 # v5.2.0
        with:
          java-version: '21'
          distribution: 'temurin'

      - name: Cache Maven repository
        if: matrix.runner == 'java'
        uses: actions/cache@cdf6c1fa76f9f475f3d7449005a359c84ca0f306 # v5.0.3
        with:
          path: ~/.m2
          key: ${{ runner.os }}-m2-iam-${{ hashFiles('iam/java/pom.xml') }}
          restore-keys: ${{ runner.os }}-m2-iam-

      - name: Set up Python
        if: matrix.runner == 'python'
        uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065 # v5.6.0
        with:
          python-version: '3.12'

      - name: Install Python dependencies
        if: matrix.runner == 'python'
        working-directory: iam/python
        run: pip install -r requirements.txt

      - name: Start ArcadeDB
        working-directory: iam
        run: docker compose up -d

      - name: Setup database
        working-directory: iam
        run: ./setup.sh

      - name: Run curl queries
        if: matrix.runner == 'curl'
        working-directory: iam
        run: ./queries/queries.sh

      - name: Build and run Java
        if: matrix.runner == 'java'
        working-directory: iam/java
        run: |
          mvn package --no-transfer-progress
          java -jar target/iam.jar

      - name: Run Python queries
        if: matrix.runner == 'python'
        working-directory: iam/python
        run: python iam.py

      - name: Teardown
        if: always()
        working-directory: iam
        run: docker compose down
```

> **Note:** Pin the `actions/setup-python` SHA. Verify the latest pinned SHA from another workflow or from GitHub.

```bash
git commit -m "ci: add IAM workflow with curl, java, and python runners"
```
